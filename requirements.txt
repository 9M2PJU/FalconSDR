# PySide6, numpy, scipy and pyqtgraph all have reliable prebuilt wheels on
# every platform (Linux/Windows/macOS) - pip install these directly, no need
# for apt/system packages:
PySide6>=6.6
numpy>=1.26
scipy>=1.11
pyqtgraph>=0.13
# Windows has no system time zone database - zoneinfo needs this package
# there (on Linux/macOS it's harmless, the system one is used first).
tzdata
# RTL-SDR without SoapySDR: pyrtlsdrlib ships a prebuilt librtlsdr (incl.
# RTL-SDR Blog V4 support) for Windows/macOS/Linux, so on Windows/macOS the
# dongle works with pip alone. On Windows the dongle still needs the WinUSB
# driver (Zadig) - the same one SDR++/SDR# already need.
pyrtlsdr>=0.3
pyrtlsdrlib
# Live audio on Windows/macOS: wheels bundle PortAudio, no compiler needed.
# Used only when PyAudio (below) isn't installed - Linux keeps PyAudio.
sounddevice
#
# SoapySDR (+ device drivers, e.g. SoapySDRRTLSDR) and PyAudio/PortAudio are
# the two exceptions - they need system-level installation:
#   Linux (Ubuntu/Debian):  sudo apt install python3-soapysdr soapysdr0.8-module-rtlsdr python3-pyaudio
#   Windows:                nothing extra - RTL-SDR via pyrtlsdr, audio via
#                           sounddevice (both above); SoapySDR optional
#   macOS:                  nothing extra, same as Windows; SoapySDR optional
#                           (brew install soapysdr soapyrtlsdr)
# If no SDR library is available at all, the app still starts and falls back
# to the built-in Simulated backend rather than failing.
#
# QtMultimedia (used by the Odtwarzacz tab's audio player) is bundled in the
# single PySide6 wheel - no separate package needed, unlike PyQt6/apt.
#
# --- Optional: MP3 recording output (Options tab -> Recording -> MP3) ------
# Requires a system `ffmpeg` on PATH - not a pip package (same reasoning as
# SoapySDR/PyAudio above). If it's missing, the app still starts fine and
# just keeps writing WAV regardless of the MP3 setting (see
# smartscan/audio_transcode.py):
#   Linux (Ubuntu/Debian):  sudo apt install ffmpeg
#   Windows:                winget install ffmpeg  (or download a build and add it to PATH)
#   macOS:                  brew install ffmpeg
#
# --- Optional: transcription (Odtwarzacz tab) -------------------------------
# faster-whisper (and its dependencies: ctranslate2, av, tokenizers...) is NOT
# installed into the system Python - Ubuntu 24.04 blocks that (PEP 668), and
# it's also fairly heavy (ASR models), so it lives in its own venv instead:
#
#   python3 -m venv --system-site-packages .venv
#   .venv/bin/pip install faster-whisper
#   # optional, if you want GPU (NVIDIA/CUDA) transcription:
#   .venv/bin/pip install nvidia-cublas-cu12 nvidia-cudnn-cu12
#
# --system-site-packages means the venv still sees PySide6/numpy, so the rest
# of the app works the same. main.py switches into this venv itself at
# startup (if it exists), so running "python3 main.py" doesn't change.
# Whisper's models (large, ~1.5 GB + ~0.8 GB) download themselves from
# Hugging Face on first transcription and land in ~/.cache/huggingface.
faster-whisper>=1.2
